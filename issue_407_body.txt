use std::collections::HashSet;

pub struct SearchResult {
    pub rowid: i64,
    pub score: f32,
}

/// Merges FTS results and Semantic results using Reciprocal Rank Fusion (RRF)
pub fn reciprocal_rank_fusion(
    fts_results: &[SearchResult],
    semantic_results: &[SearchResult],
    k: usize,
) -> Vec<SearchResult> {
    let mut scores = std::collections::HashMap::new();
    let rrf_k = 60.0; // standard constant

    for (rank, result) in fts_results.iter().enumerate() {
        let score = 1.0 / (rrf_k + (rank as f32 + 1.0));
        *scores.entry(result.rowid).or_insert(0.0) += score;
    }

    for (rank, result) in semantic_results.iter().enumerate() {
        let score = 1.0 / (rrf_k + (rank as f32 + 1.0));
        *scores.entry(result.rowid).or_insert(0.0) += score;
    }

    let mut merged: Vec<SearchResult> = scores
        .into_iter()
        .map(|(rowid, score)| SearchResult { rowid, score })
        .collect();

    merged.sort_by(|a, b| b.score.partial_cmp(&a.score).unwrap_or(std::cmp::Ordering::Equal));
    merged.truncate(k);
    merged
}

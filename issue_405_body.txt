use candle_core::{Device, Tensor, Result};

pub struct GpuCosineCache {
    embeddings: Tensor,
    device: Device,
}

impl GpuCosineCache {
    pub fn new(embeddings: Tensor) -> Self {
        Self {
            device: embeddings.device().clone(),
            embeddings,
        }
    }

    pub fn compute_similarities(&self, query: &Tensor) -> Result<Tensor> {
        // 1. L2 Normalize the query
        let q_norm = query.sqr()?.sum_all()?.sqrt()?;
        let query_normalized = query.broadcast_div(&q_norm)?;

        // 2. L2 Normalize the cached embeddings (assuming they aren't pre-normalized)
        let cache_norm = self.embeddings.sqr()?.sum_keepdim(1)?.sqrt()?;
        let cache_normalized = self.embeddings.broadcast_div(&cache_norm)?;

        // 3. Matrix multiplication (Cosine Similarity = Q_norm * Cache_norm^T)
        // Ensure dims match for matmul
        cache_normalized.matmul(&query_normalized.transpose(0, 1)?)
    }
}

```rust
use std::collections::HashMap;

pub struct ActiveVramCache {
    /// Maps SQLite rowid to Tensor row index
    pub rowid_to_idx: HashMap<i64, usize>,
    /// Maps Tensor row index to SQLite rowid
    pub idx_to_rowid: HashMap<usize, i64>,
}

impl ActiveVramCache {
    pub fn new() -> Self {
        Self {
            rowid_to_idx: HashMap::new(),
            idx_to_rowid: HashMap::new(),
        }
    }

    pub fn insert_mapping(&mut self, rowid: i64, idx: usize) {
        self.rowid_to_idx.insert(rowid, idx);
        self.idx_to_rowid.insert(idx, rowid);
    }

    pub fn get_rowid(&self, idx: usize) -> Option<i64> {
        self.idx_to_rowid.get(&idx).copied()
    }
}
```
